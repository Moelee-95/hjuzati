function img_hover(element){
	var thId = element.id;
	var imge = thId +"_img";
	document.getElementById(thId).style.display = "block";
	document.getElementById(imge).style.opacity ="0.3";

}
function img_unhover(element){
	var thId = element.id;
	var imge = thId +"_img";
	document.getElementById(thId).style.display = "none";
	document.getElementById(imge).style.opacity ="1";

}
